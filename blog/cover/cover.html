mqttkit

Elysia-style application framework for MQTT — ordered middleware, typed topic routes, MQTT 5 RPC, AsyncAPI docs.
mqttkit.keyp.dev · @mqttkit/core · MIT
app.ts 
const app = new MqttApp()
  .use(aedes({ tcp: { port: 1883 } }))
  .use(
    router().topic('devices/:uid/events', {
      publish: ({ params, principal }) =>
        params.uid === principal?.uid,
      schema: { body: z.object({ temperature: z.number() }) },
      async onMessage(ctx) {
        // ctx.body.temperature  → number ✓
        await ctx.publish(`server/${ctx.params.uid}/ack`, 'ok')
      },
    }),
  )

await app.listen()
